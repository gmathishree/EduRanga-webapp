  const maxWrongGuesses = 6;
  let selected = null;
  let guessedLetters = [];
  let wrongGuesses = 0;
  let questionIndex = 0;
  let currentQuestions = [];
  let score = 0;
  let gamesPlayed = 0;
  let badges = new Set();

  const wordDisplay = document.getElementById("wordDisplay");
  const hintDisplay = document.getElementById("hint");
  const lettersDiv = document.getElementById("letters");
  const messageDiv = document.getElementById("message");
  const hangmanParts = [
    document.getElementById("head"),
    document.getElementById("body"),
    document.getElementById("leftArm"),
    document.getElementById("rightArm"),
    document.getElementById("leftLeg"),

  const selectionContainer = document.getElementById("selectionContainer");
  const gameContainer = document.getElementById("gameContainer");
  const leaderboardDiv = document.getElementById("leaderboard");
  const badgeDisplay = document.getElementById("badgeDisplay");

  const startBtn = document.getElementById("startBtn");
  const nextBtn = document.getElementById("nextBtn");
  const endBtn = document.getElementById("endBtn");

  startBtn.addEventListener("click", () => {
    questionsSetup();
    selectionContainer.style.display = "none";
    gameContainer.style.display = "block";
    animateMessage("");
    score = 0;
    questionIndex = 0;
    badges.clear();
    showNextWord();
  });

  nextBtn.addEventListener("click", () => {
    questionIndex++;
    if(questionIndex < currentQuestions.length) {
      showNextWord();
    } else {
      endGame();
    }
  });

  endBtn.addEventListener("click", () => {
    endGame();
    setTimeout(() => {
      // Redirect to the login.html page (make sure this path is correct for your setup)
      window.location.href = "final_HORIZON.html";
    }, 1500); // delay for message display
  });

  function questionsSetup() {
    const difficulty = difficultySelect.value;
    const subject = subjectSelect.value.toLowerCase();
    const subjectCategory = wordsData[subject] || wordsData.general;
    const wordList = subjectCategory[difficulty] || subjectCategory.easy;

    // Shuffle and take minimum 5 words or available count
    currentQuestions = shuffleArray(wordList).slice(0, 5);
  }

  function showNextWord() {
    guessedLetters = [];
    wrongGuesses = 0;
    hangmanParts.forEach(part => {
      part.style.display = "none";
      part.classList.remove("show");
    });
    animateMessage("");
    nextBtn.style.display = "none";

    selected = currentQuestions[questionIndex];
    hintDisplay.textContent = "Hint: " + selected.hint;
    animateHint();
    createLetterButtons();
    updateWordDisplay();
  }

  function createLetterButtons() {
    lettersDiv.innerHTML = "";
    for (let i = 65; i <= 90; i++) {
      let btn = document.createElement("button");
      btn.textContent = String.fromCharCode(i);
      btn.className = "letter-btn";
      btn.onclick = () => guessLetter(btn.textContent.toLowerCase(), btn);
      lettersDiv.appendChild(btn);
    }
  }

  function guessLetter(letter, btn) {
    btn.disabled = true;
    guessedLetters.push(letter);

    if (selected.word.includes(letter)) {
      btn.classList.add('correct');
      updateWordDisplay();
      checkWin();
    } else {
      btn.classList.add('wrong');
      wrongGuesses++;
      updateHangman();
      if (wrongGuesses >= maxWrongGuesses) {
        endCurrentWord(false);
      }
    }
  }

  function updateWordDisplay() {
    let display = "";
    for (let char of selected.word) {
      if (guessedLetters.includes(char)) {
        display += char.toUpperCase() + " ";
      } else {
        display += "_ ";
      }
    }
    wordDisplay.textContent = display.trim();
  }

  function updateHangman() {
    if (wrongGuesses > 0 && wrongGuesses <= maxWrongGuesses) {
      animateHangmanPart(wrongGuesses - 1);
    }
  }

  function checkWin() {
    let won = selected.word.split("").every(letter => guessedLetters.includes(letter));
    if (won) {
      endCurrentWord(true);
    }
  }

  function endCurrentWord(won) {
    lettersDiv.querySelectorAll("button").forEach(btn => btn.disabled = true);
    if (won) {
      animateMessage("Correct! Well done!");
      score += 10;
      checkBadges();
    } else {
      animateMessage(`Oops! The word was: ${selected.word.toUpperCase()}`);
    }
    nextBtn.style.display = questionIndex < currentQuestions.length - 1 ? "inline-block" : "none";
  }

  function endGame() {
    lettersDiv.innerHTML = "";
    nextBtn.style.display = "none";
    animateMessage(`Game Over! Score: ${score}`);
    updateLeaderboard(score);
    showBadges();
    gameContainer.style.display = "none";
    selectionContainer.style.display = "block";
  }

  function updateLeaderboard(newScore) {
    gamesPlayed++;
    // Save in localStorage
    let leaderboard = JSON.parse(localStorage.getItem("hangmanLeaderboard") || "[]");
    leaderboard.push({ score: newScore, date: new Date().toLocaleString() });
    leaderboard.sort((a,b) => b.score - a.score);
    if(leaderboard.length > 10) leaderboard.length = 10; // Keep top 10
    localStorage.setItem("hangmanLeaderboard", JSON.stringify(leaderboard));
    displayLeaderboard(leaderboard);
  }

  function displayLeaderboard(data) {
    if(data.length === 0) {
      leaderboardDiv.textContent = "No games played yet.";
      return;
    }
    let html = "<ol>";
    data.forEach(entry => {
      html += `<li>Score: ${entry.score} - ${entry.date}</li>`;
    });
    html += "</ol>";
    leaderboardDiv.innerHTML = html;
  }

  function checkBadges() {
    // Example badges based on score milestones
    if(score >= 50) badges.add("🏅 Scored 50+");
    if(score >= 80) badges.add("🎖️ Scored 80+");
    if(score === currentQuestions.length * 10) badges.add("🏆 Perfect Score");
  }

  function showBadges() {
    if(badges.size === 0) {
      badgeDisplay.textContent = "None yet.";
      return;
    }
    badgeDisplay.innerHTML = "";
    badges.forEach(badge => {
      const span = document.createElement("span");
      span.className = "badge";
      span.title = badge;
      span.textContent = badge.split(" ")[0]; // Show emoji only
      badgeDisplay.appendChild(span);
      // Trigger animation
      span.style.animationDelay = "0.3s";
    });
  }

  // Animation helper functions
  function animateMessage(text) {
    messageDiv.textContent = text;
    messageDiv.classList.remove("show");
    void messageDiv.offsetWidth; // Trigger reflow for restart animation
    messageDiv.classList.add("show");
  }

  function animateHint() {
    hintDisplay.classList.remove("show");
    void hintDisplay.offsetWidth;
    hintDisplay.classList.add("show");
  }

  function animateHangmanPart(index) {
    if(index >= 0 && index < hangmanParts.length) {
      const part = hangmanParts[index];
      part.style.display = "block";
      part.classList.remove("show");
      void part.offsetWidth;
      part.classList.add("show");
    }
  }

  // Utility shuffle function
  function shuffleArray(arr) {
    let array = arr.slice();
    for(let i = array.length - 1; i > 0; i--){
      const j = Math.floor(Math.random() * (i + 1));
      [array[i], array[j]] = [array[j], array[i]];
    }
    return array;
  }

      }, index * 500); // 0ms, 500ms, 1000ms delay sequentially
    });
    let leaderboard = JSON.parse(localStorage.getItem("hangmanLeaderboard") || "[]");
    displayLeaderboard(leaderboard);
  });
