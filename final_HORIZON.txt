<script>
let currentUserType = '';
let currentStudent = null;
let currentGame = '';

// LANDING PAGE
function selectUserType(type){
  currentUserType = type;
  document.getElementById('landingPage').classList.add('hidden');
  if(type === 'student'){
    document.getElementById('studentAuthPage').classList.remove('hidden');
    document.getElementById('authTitle').innerText = 'Student Registration';
    document.getElementById('authButton').innerText = 'Register';
  } else {
    document.getElementById('teacherAuthPage').classList.remove('hidden');
  }
}

// STUDENT AUTH
function switchToLogin(){
  document.getElementById('authTitle').innerText = 'Student Login';
  document.getElementById('authButton').innerText = 'Login';
  document.getElementById('switchAuth').innerHTML = `Don't have an account? <a href="#" onclick="switchToRegister()">Register</a>`;
}
function switchToRegister(){
  document.getElementById('authTitle').innerText = 'Student Registration';
  document.getElementById('authButton').innerText = 'Register';
  document.getElementById('switchAuth').innerHTML = `Already have an account? <a href="#" onclick="switchToLogin()">Login</a>`;
}
function registerStudent(){
  const displayName = document.getElementById('displayName').value.trim();
  const username = document.getElementById('username').value.trim();
  const password = document.getElementById('password').value.trim();
  if(!displayName || !username || !password){ alert('All fields required'); return; }
  let students = JSON.parse(localStorage.getItem('students') || '{}');
  if(document.getElementById('authButton').innerText === 'Register'){
    if(students[username]) { alert('Username already exists'); return; }
    students[username] = {displayName, password};
    localStorage.setItem('students', JSON.stringify(students));
    alert('Registration successful! Please login.');
    switchToLogin();
  } else {
    if(!students[username] || students[username].password !== password){ alert('Invalid credentials'); return; }
    currentStudent = {username, displayName: students[username].displayName};
    showStudentDashboard();
  }
}
function showStudentDashboard(){
  document.getElementById('studentAuthPage').classList.add('hidden');
  document.getElementById('studentDashboard').classList.remove('hidden');
  document.getElementById('studentDisplayName').innerText = currentStudent.displayName;
}

// TEACHER AUTH
function loginTeacher(){
  const username = document.getElementById('teacherUsername').value.trim();
  const password = document.getElementById('teacherPassword').value.trim();
  if(username === 'admin' && password === 'admin123'){
    showTeacherDashboard();
  } else {
    alert('Invalid teacher credentials');
  }
}
function showTeacherDashboard(){
  document.getElementById('teacherAuthPage').classList.add('hidden');
  document.getElementById('teacherDashboard').classList.remove('hidden');
  updateTeacherLeaderboard();
}

// START GAME
function startGame(game){
  currentGame = game;
  localStorage.setItem('currentUser', currentStudent.username);
  localStorage.setItem('currentGame', game);
  if(game === 'board_quiz') {
    window.open('snake_final.html', '_blank');
  } else if(game === 'infinity') {
    window.open('pg2.html', '_blank');
  } else if(game === 'hangman') {
    window.open('hangman_ani2.html', '_blank');
  } else if(game === 'fill_the_formula') {
    window.open('ftf333.html', '_blank');
  } else {
    window.open(`${game}_full.html`, '_blank');
  }
}

// STUDENT LEADERBOARD
function showLeaderboard(){
  document.getElementById('studentDashboard').classList.add('hidden');
  document.getElementById('leaderboardUser').innerText = currentStudent.displayName;
  document.getElementById('leaderboardPage').classList.remove('hidden');
  let leaderboard = JSON.parse(localStorage.getItem('leaderboard') || '{}');
  let userScores = leaderboard[currentStudent.username] || {};
  let tbody = '';
  for(let g in userScores){
    let s = userScores[g];
    let badge = s >= 90 ? 'Gold' : s >= 70 ? 'Silver' : 'Bronze';
    let badgeClass = s >= 90 ? 'gold' : s >= 70 ? 'silver' : 'bronze';
    let suggestion = s >= 90 ? 'Excellent! Keep it up.' :
                   s >= 70 ? 'Good! A little more practice can make it perfect.' :
                   s >= 50 ? 'Fair. Try reviewing the key concepts.' :
                            'Needs improvement. Focus on fundamentals.';
    tbody += `<tr>
      <td>${g.replace(/_/g,' ')}</td>
      <td>${s}</td>
      <td><span class="badge ${badgeClass}">${badge}</span></td>
      <td>${suggestion}</td>
    </tr>`;
  }
  document.getElementById('leaderboardBody').innerHTML = tbody;
}
function backToDashboard(){
  document.getElementById('leaderboardPage').classList.add('hidden');
  document.getElementById('studentDashboard').classList.remove('hidden');
}

// TEACHER LEADERBOARD
function updateTeacherLeaderboard(){
  let leaderboard = JSON.parse(localStorage.getItem('leaderboard') || '{}');
  let studentsData = JSON.parse(localStorage.getItem('students') || '{}');
  let tbody = '';
  for(let username in leaderboard){
    let displayName = studentsData[username] ? studentsData[username].displayName : username;
    let games = leaderboard[username];
    for(let game in games){
      let score = games[game];
      let badge = score >= 90 ? 'Gold' : score >= 70 ? 'Silver' : 'Bronze';
      let badgeClass = score >= 90 ? 'gold' : score >= 70 ? 'silver' : 'bronze';
      let suggestion = score >= 90 ? 'Excellent! Keep it up.' :
                     score >= 70 ? 'Good! A little more practice can make it perfect.' :
                     score >= 50 ? 'Fair. Try reviewing the key concepts.' :
                                  'Needs improvement. Focus on fundamentals.';
      tbody += `<tr>
        <td>${displayName}</td>
        <td>${game.replace(/_/g,' ')}</td>
        <td>${score}</td>
        <td><span class="badge ${badgeClass}">${badge}</span></td>
        <td>${suggestion}</td>
      </tr>`;
    }
  }
  document.getElementById('teacherLeaderboardBody').innerHTML = tbody;
}

// LOGOUT
function logout(){
  currentStudent = null;
  currentUserType = '';
  currentGame = '';
  document.getElementById('landingPage').classList.remove('hidden');
  document.getElementById('studentAuthPage').classList.add('hidden');
  document.getElementById('studentDashboard').classList.add('hidden');
  document.getElementById('teacherAuthPage').classList.add('hidden');
  document.getElementById('teacherDashboard').classList.add('hidden');
  document.getElementById('leaderboardPage').classList.add('hidden');
  document.getElementById('displayName').value = '';
  document.getElementById('username').value = '';
  document.getElementById('password').value = '';
  document.getElementById('teacherUsername').value = '';
  document.getElementById('teacherPassword').value = '';
}

// LISTEN FOR GAME SCORE UPDATES
window.addEventListener('storage', function(e){
  if(e.key === 'latestScore' && currentStudent){
    const score = parseInt(e.newValue);
    let leaderboard = JSON.parse(localStorage.getItem('leaderboard') || '{}');
    if(!leaderboard[currentStudent.username]) leaderboard[currentStudent.username] = {};
    leaderboard[currentStudent.username][localStorage.getItem('currentGame')] = score;
    localStorage.setItem('leaderboard', JSON.stringify(leaderboard));
    localStorage.removeItem('latestScore');
    showLeaderboard();
  }
});
