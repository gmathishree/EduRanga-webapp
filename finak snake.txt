function createBoard() {
  board.innerHTML = '';
  positionToIndexMap = [];
  let leftToRight = true;
  for(let r=9; r>=0; r--) {
    for(let c=0; c<10; c++) {
      let pos = r*10 + (leftToRight ? c+1 : 10-c);
      const cell = document.createElement('div');
      cell.className = 'cell';
      cell.textContent = pos;
      if(ladders[pos]) cell.classList.add('ladderStart');
      if(snakes[pos]) cell.classList.add('snakeHead');
      if(quizTiles.includes(pos)) cell.classList.add('quizTile');
      board.appendChild(cell);
      positionToIndexMap[pos] = board.children.length -1;
    }
    leftToRight = !leftToRight;
  }
}

function moveTokenTo(pos) {
  if(pos < 1) pos = 1;
  if(pos > 100) pos = 100;
  const idx = positionToIndexMap[pos];
  if(typeof idx === 'undefined' || idx === null) {
    console.error('Invalid positionToIndexMap lookup:', pos);
    return;
  }
  const cell = board.children[idx];
  if(!cell) {
    console.error('No cell found for position:', pos, 'index:', idx);
    return;
  }
  const containerRect = boardContainer.getBoundingClientRect();
  const rect = cell.getBoundingClientRect();
  const left = rect.left - containerRect.left + rect.width/2 - token.offsetWidth/2;
  const top = rect.top - containerRect.top + rect.height/2 - token.offsetHeight/2;
  token.style.left = `${left}px`;
  token.style.top = `${top}px`;
  position = pos;
  console.log('Moved token to position:', pos);
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function animateTokenPositions(startPos, endPos) {
  if(startPos === endPos) return;
  const step = (endPos > startPos) ? 1 : -1;
  for(let i = startPos + step; (step > 0 ? i <= endPos : i >= endPos); i += step) {
    moveTokenTo(i);
    await sleep(300);
  }
}

async function animateMove(steps) {
  if(animating) return;
  animating = true;
  try {
    let oldPos = position;
    let targetPos = position + steps;
    if(targetPos > 100) targetPos = 100;
    console.log(`Animating move from ${oldPos} to ${targetPos}`);
    await animateTokenPositions(oldPos, targetPos);
    position = targetPos;

    if(ladders[position]) {
      message.textContent = `Ladder! Move up to ${ladders[position]}`;
      await sleep(800);
      await animateTokenPositions(position, ladders[position]);
      position = ladders[position];
    }
    else if(snakes[position]) {
      message.textContent = `Snake! Slide down to ${snakes[position]}`;
      await sleep(800);
      await animateTokenPositions(position, snakes[position]);
      position = snakes[position];
    }

    if(quizTiles.includes(position)) {
      await askQuestion(position);
    }

    if(position === 100) {
      message.textContent = '🎉 You have won! Congratulations!';
      dice.style.pointerEvents = 'none';
    } else {
      message.textContent = 'Roll the dice!';
      dice.style.pointerEvents = 'auto';
    }
  } catch(e) {
    console.error('Error in animateMove:', e);
    message.textContent = 'Error occurred! You can roll again.';
    dice.style.pointerEvents = 'auto';
  } finally {
    animating = false;
  }
}

async function askQuestion(pos) {
  if(!selectedGrade) return;
  const quizSet = questions[selectedGrade];
  const q = quizSet && quizSet[pos];
  if(!q) {
    console.warn('Missing question for position', pos);
    return;
  }
  return new Promise((resolve) => {
    qText.textContent = q.q;
    qOptions.innerHTML = '';
    q.options.forEach((opt, idx) => {
      const btn = document.createElement('button');
      btn.className = 'answer-btn';
      btn.textContent = opt;
      btn.onclick = () => {
        Array.from(qOptions.children).forEach(b => b.disabled = true);
        if(idx === q.answer){
          btn.classList.add('correct');
          message.textContent = 'Correct! Advance 1 step.';
          animateTokenPositions(position, Math.min(position+1, 100)).then(() => {
            position = Math.min(position+1, 100);
            updateAfterQuestion();
          });
        } else {
          btn.classList.add('wrong');
          message.textContent = 'Wrong! Move back 1 step.';
          animateTokenPositions(position, Math.max(position-1, 1)).then(() => {
            position = Math.max(position-1, 1);
            updateAfterQuestion();
          });
        }
        setTimeout(() => {
          questionModal.style.display = 'none';
          resolve();
        }, 1500);
      };
      qOptions.appendChild(btn);
    });
    questionModal.style.display = 'flex';
    dice.style.pointerEvents = 'none';
  });
}

function updateAfterQuestion() {
  message.textContent = 'Roll the dice!';
  dice.style.pointerEvents = 'auto';
}

async function rollDice() {
  if(animating) return;
  if(position === 100){
    message.textContent = 'Game complete. Refresh to play again.';
    dice.style.pointerEvents = 'none';
    return;
  }
  dice.style.pointerEvents = 'none';
  message.textContent = 'Rolling...';
  for(let i=0; i<12; i++) {
    const val = Math.floor(Math.random()*6) + 1;
    dice.className = 'dice-' + val;
    await sleep(70);
  }
  const roll = Math.floor(Math.random()*6) + 1;
  dice.className = 'dice-' + roll;
  message.textContent = `You rolled ${roll}`;
  await sleep(350);
  await animateMove(roll);
}

dice.addEventListener('click', rollDice);

  selectedGrade = gradeSelect.value;
  position = 1;
  animating = false;
  createBoard();
  moveTokenTo(position);
  dice.removeAttribute('disabled');
  message.textContent = 'Click dice to roll!';
});
